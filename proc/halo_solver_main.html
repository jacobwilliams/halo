    subroutine halo_solver_main(config_file_name,debug,gui,iteration_callback,stop_check,traj_file,epoch_callback)

    use parameters_module
!$  use omp_lib

    implicit none

    character(len=*),intent(in) :: config_file_name  !! the config file to read
    logical,intent(in) :: debug !! for debugging prints
    logical,intent(in),optional :: gui !! run for the GUI (`halo_gui`): the JSON trajectory
                                       !! file is always exported, and the pyvista script
                                       !! is not run. [Default is False]
    procedure(iteration_callback_func),optional :: iteration_callback
        !! if present, it receives the trajectory of each solver iteration
        !! (in memory: no files are written for them). Note: this
        !! propagates all the segments at each iteration.
    procedure(stop_check_func),optional :: stop_check
    procedure(epoch_callback_func),optional :: epoch_callback
        !! if present, it receives the epochs at the constraint points of each
        !! solver iteration (cheap: no propagation)
        !! if present, it is checked at each solver iteration, and the
        !! solver stops if it returns true
    character(len=:),allocatable,intent(out),optional :: traj_file
        !! the JSON trajectory file written for the solution (if any)

    type(my_solver_type),target :: solver  !! an instance of the solver that we will use
                                           !! (target: [[eclipse_optimize]] points to it)
    real(wp),dimension(:),allocatable :: x  !! solver opt vars vector ["forward-backward" formulation]
    real(wp),dimension(:),allocatable :: x_guess  !! the initial guess (for `generate_solution_plots`)
    character(len=:),allocatable :: plot_data_file  !! for `generate_solution_plots`
    integer :: istat_plot  !! exit status of the plot script
    integer :: m  !! number of functions
    real(wp),dimension(:),allocatable :: f  !! function vector (constraint violations)
    real(wp) :: tstart, tend  !! for timing
    real(wp) :: tstart_cpu, tend_cpu  !! for timing
    integer :: istat
    character(len=:),allocatable :: message  !! Text status message from solver
    integer :: n_segs, iseg
    real(wp),dimension(6) :: x_rotating
    character(len=:),allocatable :: mkspk_input, bsp_output, mkspk_setup, pck_output  !! filenames for mkspk
!$  integer :: tid, nthreads

    write(*,'(A)') ''
    write(*,'(A)') ' * HALO start'

!$OMP PARALLEL PRIVATE(NTHREADS, TID)
!$
!$  tid = omp_get_thread_num()
!$
!$  if (tid == 0) then
!$      nthreads = omp_get_num_threads()
!$      write(*,'(A,1X,I3)') ' * Number of OMP threads: ', OMP_get_num_threads()
!$  end if
!$
!$OMP END PARALLEL

    if (debug) then
        write(*,*) ''
        write(*,*) '----------------------'
        write(*,*) 'Initializing...'
        write(*,*) '----------------------'
        write(*,*) ''
    end if

    call solver%init(config_file_name,x)  ! initialize the solver & mission (and generate the initial guess)

    if (present(gui)) then
        if (gui) then
            ! the GUI plots the JSON trajectory file itself
            solver%mission%generate_json_trajectory_file = .true.
            solver%mission%run_pyvista_script = .false.
        end if
    end if
    if (present(iteration_callback)) solver%iteration_callback => iteration_callback
    if (present(stop_check)) solver%stop_check => stop_check  ! (see halo_user_input_check)
    if (present(epoch_callback)) solver%epoch_callback => epoch_callback

    if (solver%mission%generate_plots) &
        call solver%mission%plot('guess', draw_trajectory=.true.)    ! plot the initial guess
    if (solver%mission%generate_guess_and_solution_files) &
        call solver%mission%write_optvars_to_file('guess',x)    ! write guess to a file

    !....debugging....
    ! if (solver%mission%generate_trajectory_files) &
    !     call solver%mission%plot('guess',&
    !             draw_trajectory = .false., &
    !             export_trajectory=solver%mission%generate_trajectory_files)
    !....debugging....

    if (debug) then
        call solver%mission%define_problem_size(m=m)
        allocate(f(m))
        call solver%mission%constraint_violations(x,f)
        write(*,*) ''
        write(*,*) '----------------------'
        write(*,*) 'Initial Guess...'
        write(*,*) '----------------------'
        write(*,*) ''
        write(*,'(A/,*(F30.16/))') 'x:      ', x * solver%mission%xscale + solver%mission%xoffset  ! unscaled values
        write(*,*) ''
        write(*,'(A/,*(F30.16/))') 'f:      ', f    ! scaled values
        write(*,*) ''
    end if

    if (debug) then
        write(*,*) 'INITIAL GUESS:'
        call solver%mission%define_problem_size(n_segs=n_segs)
        do iseg = 1, n_segs
            call solver%mission%segs(iseg)%get_inputs(x0_rotating=x_rotating)
            write(*,'(I5, *(F15.6,1X))') iseg, x_rotating
        end do
    end if

    if (solver%mission%solve) then
        if (debug) then
            write(*,*) ''
            write(*,*) '----------------------'
            write(*,*) 'Solving...'
            write(*,*) '----------------------'
            write(*,*) ''
        end if
        write(*,'(A)') ' * Solving'
        write(*,*) ''

        if (solver%mission%generate_solution_plots) x_guess = x  ! (for the plots)
        call cpu_time(tstart_cpu)
!$      tstart = omp_get_wtime()
        if (solver%mission%sqpopt_solve) then  ! TEMPORARY experiment
            call solver%eclipse_optimize(x)
            message = 'solved with sqpopt (sqpopt_solve)'
        else
        call solver%solve(x)  ! call the solver
        call solver%status(istat=istat, message=message)
        end if
        call cpu_time(tend_cpu)
!$      tend = omp_get_wtime()
        call solver%mission%put_x_in_segments(x) ! populate segs with solution
        if (solver%mission%eclipse_optimization .and. .not. solver%mission%sqpopt_solve) then
            call solver%eclipse_optimize(x)  ! move it out of the Earth's shadow
            call solver%mission%put_x_in_segments(x)
!$          tend = omp_get_wtime()
            call cpu_time(tend_cpu)
            message = message//' (then eclipse optimization)'
        end if
    else if (solver%mission%eclipse_optimization) then
        ! the initial guess is taken as the solution: only the eclipse optimization
        call cpu_time(tstart_cpu)
!$      tstart = omp_get_wtime()
        call solver%mission%put_x_in_segments(x)
        call solver%eclipse_optimize(x)
        call solver%mission%put_x_in_segments(x)
!$      tend = omp_get_wtime()
        call cpu_time(tend_cpu)
        message = 'Not solved (eclipse optimization of the initial guess)'
    else
        call cpu_time(tstart_cpu)
        call cpu_time(tend_cpu)
        message = 'Not solved'
    end if

    if (debug) then
        write(*,*) ''
        write(*,*) '----------------------'
        write(*,*) 'Solution...'
        write(*,*) '----------------------'
        write(*,*) ''
    end if

    write(*,*) ''
    write(*,'(A)') ' * Status: '//message
    write(*,'(A,1x,F10.3,1x,a)') ' * Elapsed cpu_time: ', (tend_cpu-tstart_cpu), 'sec'
!$  write(*,'(A,1x,F10.3,1x,a)') ' * OMP wall time   : ', (tend-tstart), 'sec'

    if (solver%mission%generate_guess_and_solution_files) then
        write(*,'(A)') ' * Generate solution file'
        call solver%mission%write_optvars_to_file('solution',x) ! write solution to a file
    end if

    if (solver%mission%generate_solution_plots .and. allocated(x_guess)) then
        plot_data_file = 'solution_plot_data_'//solver%mission%get_case_name()//'.json'
        write(*,'(A)') ' * Solution plot data: '//plot_data_file
        call solver%mission%write_solution_plot_data(plot_data_file, x_guess, x)
        write(*,'(A)') ' * Solution plots (python/plot_solution.py)'
        call execute_command_line('python ./python/plot_solution.py '//plot_data_file, &
                                  wait=.true., exitstat=istat_plot)
        if (istat_plot /= 0) write(*,'(A)') '   (the plot script failed: it needs Python with numpy and matplotlib)'
    end if

    ! export solution to plot or trajectory file
    if (solver%mission%generate_trajectory_files .or. solver%mission%generate_plots) then
        write(*,'(A)') ' * Export solution trajectory'
        call solver%mission%plot('solution',&
                draw_trajectory=solver%mission%generate_plots, &
                export_trajectory=solver%mission%generate_trajectory_files)
    end if

    if (solver%mission%generate_kernel) then
        if (.not. solver%mission%generate_trajectory_files) then
            error stop 'error: kernel generation requires the trajectory file to be exported'
        else
            write(*,*) '* Generate BSP kernel'

            mkspk_input = 'solution_'//solver%mission%get_case_name()//'.txt'
            bsp_output  = 'solution_'//solver%mission%get_case_name()//'.bsp'
            mkspk_setup = 'mkspk_'//solver%mission%get_case_name()//'.txt'
            pck_output  = 'solution_'//solver%mission%get_case_name()//'.tpc.json'

            !note: have to delete the kernel if it is already there or mkspk will fail.
            call delete_file(bsp_output)

            call generate_mkspk_setup_file(mkspk_setup)
            call generate_pck_file(pck_output)
            call execute_command_line(mkspk_path//' -setup '//mkspk_setup//' -input '//mkspk_input//' -output '//bsp_output)
        end if
    end if

    if (debug) then
        write(*,*) 'SOLUTION:'
        call solver%mission%define_problem_size(n_segs=n_segs)
        do iseg = 1, n_segs
            call solver%mission%segs(iseg)%get_inputs(x0_rotating=x_rotating)
            write(*,'(I5, *(F15.6,1X))') iseg, x_rotating
        end do
    end if

    if (solver%mission%generate_defect_file) then
        write(*,'(A)') ' * Generate defect file'
        call solver%mission%print_constraint_defects('defects_'//&
                                                     solver%mission%get_case_name()//&
                                                     '.csv')
    end if

    if (solver%mission%generate_json_trajectory_file) then
        call solver%mission%export_trajectory_json_file('traj_'//solver%mission%get_case_name())
        if (present(traj_file)) traj_file = 'traj_'//solver%mission%get_case_name()//'.json'
        ! note: halo_gui reads the file name from this line
        write(*,'(A)') ' * Trajectory file: traj_'//solver%mission%get_case_name()//'.json'
    end if

    if (solver%mission%generate_eclipse_files) then
        call solver%mission%generate_eclipse_data('eclipse', &
                                                  filetype = solver%mission%eclipse_filetype)
    end if

    if (solver%mission%run_pyvista_script) then
        write(*,'(A)') ' * Run PyVista script'
        !mkspk_input = 'solution_'//solver%mission%get_case_name()//'.txt'
        mkspk_input = 'traj_'//solver%mission%get_case_name()//'.json'
        call execute_command_line('python ./python/plot_utilities.py '//mkspk_input, wait=.false.)
    end if

    write(*,'(A)') ' * HALO done'
    write(*,'(A)') ''

    end subroutine halo_solver_main
