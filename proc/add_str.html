    pure subroutine add_str(me, str)

    class(text_buffer),intent(inout) :: me
    character(len=*),intent(in) :: str

    call me%reserve(len(str))
    me%s(me%n+1:me%n+len(str)) = str
    me%n = me%n + len(str)

    end subroutine add_str
