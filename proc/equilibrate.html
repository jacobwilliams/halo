equilibrate

subroutine equilibrate(d, x)

Equilibrate the continuity Jacobian at x for the optimization (eclipse_opt_equilibrate): alternately normalize its rows (weights w) and its columns (the variables' factors r, the optimization's variables being x/r), a few times, so that the rows and the columns all have 2-norms near 1. This lowers the condition number of the KKT systems of sqpopt's QP subproblems (for 100 revolutions, J's condition number goes from 8e4 to 7e2). The objective, |x/r - x0/r|^2/2, is then measured in these units.

    subroutine equilibrate(d, x)

    type(eclipse_opt_data),intent(inout) :: d
    real(wp),dimension(:),intent(in) :: x

    real(wp),dimension(:),allocatable :: jac, a, norms
    integer :: it, k

    call d%mission%compute_jacobian_stm(x, jac)
    allocate(a(size(jac)), norms(max(size(d%r), size(d%w))))
    d%r = one
    d%w = one
    do it = 1, 10
        a = jac * d%r(d%icol) * d%w(d%irow)
        norms = zero
        do k = 1, size(a)
            norms(d%irow(k)) = norms(d%irow(k)) + a(k)**2
        end do
        where (norms(1:size(d%w)) > zero) d%w = d%w / sqrt(norms(1:size(d%w)))
        a = jac * d%r(d%icol) * d%w(d%irow)
        norms = zero
        do k = 1, size(a)
            norms(d%icol(k)) = norms(d%icol(k)) + a(k)**2
        end do
        where (norms(1:size(d%r)) > zero) d%r = d%r / sqrt(norms(1:size(d%r)))
    end do
    write(*,'(A,2ES10.2,A,2ES10.2)') '   equilibrated: column factors ', minval(d%r), maxval(d%r), &
                                     ', row weights ', minval(d%w), maxval(d%w)

    end subroutine equilibrate
