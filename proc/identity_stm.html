    pure function identity_stm() result(phi)

    real(wp),dimension(n_eoms*n_eoms) :: phi

    integer :: i

    phi = zero
    do i = 1, n_eoms
        phi((i-1)*n_eoms + i) = one
    end do

    end function identity_stm
